# The image library is a review board, not a picture source.
#
# Everything in assets/images/shortlist/ is a WATERMARKED ADOBE STOCK COMP —
# unlicensed, evaluation-only. The comp licence covers reviewing an image, and
# this site is reachable without a login, so the exposure worth closing is not
# someone opening the page on purpose: it is a hundred unlicensed frames being
# crawled and served up in image search to people who never came here.
#
# So the board stays usable for anyone with the link, and stays out of the index.
# images.html carries a matching <meta name="robots" content="noindex"> — this
# file stops the crawl, that stops the listing, and neither works alone.
#
# Nothing else on the site is restricted; the design system itself is meant to
# be found.

User-agent: *
Disallow: /assets/images/shortlist/
Disallow: /images.html
